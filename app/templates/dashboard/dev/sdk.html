Démarrer avec le SDK Python en 3 étapes

1 Installer la dépendance requests

Le SDK utilise uniquement requests (déjà très populaire en Python). Installez-le :

pip install requests

2 Télécharger le SDK

Clique sur le bouton "Télécharger le SDK" ci-dessus. Le fichier boost_reseller.py sera téléchargé sur ton appareil. Place-le dans le même dossier que ton script Python.

3 Écrire votre premier script

Importez le SDK et utilisez-le :

from boost_reseller import BoostReseller

# Initialisation
client = BoostReseller(api_key="bsk_live_xxxx")

# Recuperer les produits
produits = client.produits.list(limit=20)
for p in produits:
    print(f"{p['nom']} - {p['prix']} FCFA")

# Creer un produit
nouveau = client.produits.create(
    nom="T-shirt Nike",
    prix=15000,
    stock=50,
    description="T-shirt coton bio"
)
print(f"Cree : {nouveau['id']}")

# Commandes en attente
commandes = client.commandes.list(statut="en_attente")

# Mettre a jour une commande
client.commandes.update("CMD-ABC123", statut="livree")

# Voir les stats
stats = client.stats.get(periode="30j")
print(f"Revenus : {stats['revenus']} FCFA")

Exemples d'utilisation complets

from boost_reseller import BoostReseller, BoostResellerError

client = BoostReseller(api_key="bsk_live_xxxx")

# ─── BOUTIQUE ───
boutique = client.boutique.get()
print(boutique["nom"])
print(boutique["statut"])

# Mettre a jour
client.boutique.update(slogan="Nouveau slogan !")

# ─── PRODUITS ───
# Lister avec filtres
produits = client.produits.list(actif=True, limit=50)

# Creer
nouveau = client.produits.create(
    nom="Casquette",
    prix=5000,
    stock=100
)

# Modifier
client.produits.update(nouveau["id"], prix=4500, stock=80)

# Supprimer
client.produits.delete(nouveau["id"])

# ─── COMMANDES ───
commandes = client.commandes.list(statut="en_attente")
for cmd in commandes:
    print(f"{cmd['reference']} - {cmd['client_nom']} - {cmd['total']} F")

# Changer le statut
client.commandes.update("CMD-ABC123", statut="livree")

# ─── CLIENTS ───
clients = client.clients.list(limit=100)
print(f"{len(clients)} clients")

# ─── STATS ───
stats = client.stats.get(periode="30j")
print(f"Commandes : {stats['nb_commandes']}")
print(f"Revenus : {stats['revenus']} FCFA")
print(f"Clients : {stats['nb_clients']}")
print(f"Visiteurs : {stats['visiteurs']}")
print(f"Taux de conversion : {stats['taux_conversion']}%")

# ─── GESTION DES ERREURS ───
try:
    client.produits.get("id-qui-nexiste-pas")
except BoostResellerError as e:
    print(f"Erreur : {e.code} - {e}")
    print(f"HTTP : {e.status}")
